配套：面试题与答案-v3.md、面试话术-v2.md 重点四题：深拷贝、防抖节流、Promise 并发控制、数组去重。后面附了十道常见的"顺手也会问"的短题。 每题的写法：题目怎么问 → 参考实现 → 面试时怎么讲 → 追问怎么答 → 常见错误。

0. 现场写题的节奏

  1. 先问边界：输入是什么类型？要不要处理循环引用？失败要不要中断？（问一句，面试官就知道你想过）
  2. 写主逻辑，别一上来就补边界，先把主干跑通。
  3. 补边界：空值、异常、this、参数透传。
  4. 口头跑一个用例，边说边指代码。
  5. 说复杂度和取舍：时间/空间，为什么这么选。

写不出来别硬撑。说一句"我先写个基础版，边界我再补"，比卡在那里强。

1. 深拷贝

怎么问：手写一个深拷贝，考虑循环引用、Date、Map、Set。

参考实现

function deepClone(target, map = new WeakMap()) {
  // 基本类型和函数直接返回（函数一般不拷贝，直接复用）
  if (target === null || typeof target !== 'object') return target;

  // 循环引用：拷贝过的直接返回副本
  if (map.has(target)) return map.get(target);

  // 特殊对象
  if (target instanceof Date) return new Date(target);
  if (target instanceof RegExp) return new RegExp(target.source, target.flags);

  if (target instanceof Map) {
    const copy = new Map();
    map.set(target, copy);
    target.forEach((v, k) => copy.set(deepClone(k, map), deepClone(v, map)));
    return copy;
  }
  if (target instanceof Set) {
    const copy = new Set();
    map.set(target, copy);
    target.forEach(v => copy.add(deepClone(v, map)));
    return copy;
  }

  // 数组和普通对象
  const copy = Array.isArray(target)
    ? []
    : Object.create(Object.getPrototypeOf(target)); // 保留原型

  map.set(target, copy);

  for (const key of Reflect.ownKeys(target)) {
    const desc = Object.getOwnPropertyDescriptor(target, key);
    if (desc && desc.enumerable) {
      copy[key] = deepClone(target[key], map);
    }
  }
  return copy;
}

面试时怎么讲

核心就三件事：递归、区分类型、用 WeakMap 记已拷贝的对象解决循环引用。日期和正则要单独 new，Map/Set 要遍历重建，普通对象用 Object.create(原型) 保留原型链。用 WeakMap 而不是 Map，是因为它不阻止垃圾回收，拷贝完这些临时引用就能被回收。

追问

追问 答
为什么不用 JSON.parse(JSON.stringify())？ 丢函数、undefined、Symbol；Date 变字符串；RegExp 变空对象；循环引用直接报错。只适合纯 JSON 数据。
函数要不要拷贝？ 一般不拷贝，函数是行为不是数据，直接复用引用；真要拷贝得用 eval 或 new Function，不推荐。
WeakMap 和 Map 区别？ WeakMap 的 key 必须是对象、弱引用、不可遍历；这里用它正好，不阻碍 GC。
实际项目里用吗？ 用 structuredClone（浏览器原生，支持循环引用、Date、Map/Set），手写是面试考思路。
原型上的属性要拷贝吗？ 一般只拷自有可枚举属性，for...in 会把原型上的也遍历出来，所以用 Object.keys 或 Reflect.ownKeys + enumerable 判断。

常见错误

  • 忘了循环引用，碰到自己引用自己就栈溢出。
  • 只处理了数组和普通对象，Date 被当成普通对象拷成 {}。
  • for...in 把原型链上的属性也拷了。

2. 防抖 debounce

怎么问：手写防抖，支持立即执行和取消。

参考实现

function debounce(fn, wait = 300, immediate = false) {
  let timer = null;
  let result;

  function debounced(...args) {
    const callNow = immediate && !timer;   // 立即执行模式：只有第一次触发时执行
    if (timer) clearTimeout(timer);

    timer = setTimeout(() => {
      timer = null;
      if (!immediate) result = fn.apply(this, args);
    }, wait);

    if (callNow) result = fn.apply(this, args);
    return result;
  }

  debounced.cancel = () => {
    if (timer) clearTimeout(timer);
    timer = null;
  };

  return debounced;
}

用在哪（讲项目）

搜索框输入、窗口 resize 后重算表格高度、表单字段连续修改后的异步校验。育龄平台那个表格自适应就是 resize 防抖，职证宝答题卡滚动用的是节流。

追问

追问 答
防抖和节流的区别？ 防抖是"停下来才执行"，节流是"固定频率执行"。搜索框用防抖，滚动监听用节流。
immediate 什么场景用？ 按钮防连点——第一次点立刻响应，后面的连点忽略，等 wait 后再解锁。
为什么要 fn.apply(this, args)？ 保留调用时的 this 和参数，不然方法里 this 丢了。
返回值和取消防抖？ 用闭包变量 result 存最近一次的结果；cancel 挂到返回的函数上。
怎么保证最后一次一定执行？ 这就是现在的写法——每次触发都重置定时器，最后一次触发后 wait 时间到就执行。

常见错误

  • 用箭头函数包 debounced，this 指向错。
  • 忘了 clearTimeout，导致执行多次。
  • immediate 模式下没有把 timer 置空，第二次点击永远不触发。

3. 节流 throttle

怎么问：手写节流，首尾都要触发。

参考实现（时间戳 + 定时器，lodash 思路）

function throttle(fn, wait = 300, options = { leading: true, trailing: true }) {
  let last = 0;      // 上次执行时间
  let timer = null;

  return function throttled(...args) {
    const now = Date.now();

    // 第一次调用且不允许首次触发时，把 last 设成 now，避免一上来就执行
    if (!last && !options.leading) last = now;

    const remaining = wait - (now - last);

    if (remaining <= 0 || remaining > wait) {
      // 到点了（或者系统时间被改过），立即执行
      if (timer) {
        clearTimeout(timer);
        timer = null;
      }
      last = now;
      fn.apply(this, args);
    } else if (!timer && options.trailing) {
      // 没到点，但尾触发要执行：安排一次
      timer = setTimeout(() => {
        last = options.leading ? Date.now() : 0;
        timer = null;
        fn.apply(this, args);
      }, remaining);
    }
  };
}

两种简化版（面试先说这两种再说优化）

// 时间戳版：首次立即执行，最后一次可能不执行
function throttleByTime(fn, wait) {
  let last = 0;
  return function (...args) {
    const now = Date.now();
    if (now - last >= wait) {
      last = now;
      fn.apply(this, args);
    }
  };
}

// 定时器版：首次延迟执行，最后一次一定执行
function throttleByTimer(fn, wait) {
  let timer = null;
  return function (...args) {
    if (timer) return;
    timer = setTimeout(() => {
      timer = null;
      fn.apply(this, args);
    }, wait);
  };
}

面试时怎么讲

时间戳版能保证首次立即执行、但最后一次可能丢；定时器版保证最后一次执行、但首次要等。要首尾都触发，就把两者合起来：没到间隔时间时用定时器补一次尾触发，到间隔了就立即执行。lodash 的 throttle 就是这套。

追问

追问 答
滚动监听用哪个？ 节流，固定频率更新答题卡高亮或吸顶状态，不然每帧都跑太重。
为什么不用 requestAnimationFrame？ 只关心渲染帧的可以用 rAF，比 setTimeout 准；但业务上要固定间隔（比如 200ms 一次请求）还是节流。
节流能取消防抖吗？ 可以给返回函数补 cancel，跟防抖一样清定时器。
和防抖一起用？ 少见。搜索场景是防抖，滚动是节流，一般二选一。

4. Promise 并发控制

怎么问：有一批异步任务，控制同时最多跑 3 个，全部完成返回结果，保持顺序。

参考实现

/**
 * @param tasks 任务数组，每项是返回 Promise 的函数
 * @param limit 最大并发数
 */
function limitRun(tasks, limit = 3) {
  return new Promise((resolve, reject) => {
    const results = new Array(tasks.length);
    let index = 0;      // 下一个要取的任务
    let running = 0;    // 正在跑的数量
    let finished = 0;   // 已完成的数量

    if (tasks.length === 0) return resolve(results);

    const next = () => {
      // 有空位就补任务，直到并发满或者任务取完
      while (running < limit && index < tasks.length) {
        const i = index++;
        running++;

        Promise.resolve()
          .then(() => tasks[i]())     // 包一层，兼容同步任务
          .then(
            (res) => {
              results[i] = res;       // 按原始下标存，保证顺序
              running--;
              finished++;
              if (finished === tasks.length) resolve(results);
              else next();            // 空出一个位置，继续补
            },
            (err) => reject(err)      // 任意一个失败就整体失败
          );
      }
    };

    next();
  });
}

用法

const tasks = urls.map(url => () => fetch(url).then(r => r.json()));
limitRun(tasks, 3).then(results => console.log(results));

如果要求"一个失败不影响其他"

function limitRunSettled(tasks, limit = 3) {
  return new Promise((resolve) => {
    const results = new Array(tasks.length);
    let index = 0, running = 0, finished = 0;
    if (!tasks.length) return resolve(results);

    const next = () => {
      while (running < limit && index < tasks.length) {
        const i = index++;
        running++;
        Promise.resolve()
          .then(() => tasks[i]())
          .then(
            (res) => { results[i] = { status: 'fulfilled', value: res }; },
            (err) => { results[i] = { status: 'rejected', reason: err }; }
          )
          .finally(() => {
            running--; finished++;
            if (finished === tasks.length) resolve(results);
            else next();
          });
      }
    };
    next();
  });
}

面试时怎么讲

核心是一个"取任务—占坑—完成回填"的循环：维护正在运行的数量，没到上限就继续从队列里取；每个任务完成后计数减一，再补一个；结果按原始下标存，所以顺序不会乱。整个过程包在一个 Promise 里，完成数等于总数时 resolve。

追问

追问 答
为什么 results[i] = res 能保序？ 依赖的是闭包里的 i（用 const 声明在循环体内），每个任务记住自己的下标，回填时不受完成先后影响。
一个失败要不要中断？ 看业务。批量上传要 allSettled 语义（失败记下来继续）；批量提交可能要 fail-fast（有一个失败就停）。两种都实现过。
limit 能动态改吗？ 能，把 limit 放外部变量，next 每次读最新的值就行；但要注意边界（不能小于 1）。
和 Promise.all 的区别？ all 是一次性全发出去，只是等待；并发控制是限制同时在跑的数量，避免把接口打挂。
任务本身是同步函数怎么办？ Promise.resolve().then(() => task()) 包一层，同步抛错也能进 reject。
怎么加进度？ 加一个 onProgress 回调，finished 变化时调一下。

常见错误

  • 用 for 循环里 await 一个接一个，变成串行，完全没有并发。
  • 用 forEach 里 await，外层根本不等待。
  • 结果用 push，完成顺序和任务顺序不一致。

5. 数组去重

怎么问：数组去重有几种写法？对象数组怎么去重？

基本类型

// 1. Set（最简洁，O(n)）
const unique1 = arr => [...new Set(arr)];

// 2. filter + indexOf（O(n²)，NaN 会漏）
const unique2 = arr => arr.filter((item, i) => arr.indexOf(item) === i);

// 3. reduce + includes（同样 O(n²)，NaN 会漏）
const unique3 = arr => arr.reduce((acc, cur) => acc.includes(cur) ? acc : [...acc, cur], []);

// 4. Map（SameValueZero，NaN 也正确）
const unique4 = arr => {
  const map = new Map();
  arr.forEach(item => map.set(item, item));
  return [...map.values()];
};

对象数组按字段去重

function uniqueBy(arr, keyFn = item => item.id) {
  const map = new Map();
  arr.forEach(item => {
    const key = keyFn(item);
    if (!map.has(key)) map.set(key, item);  // 保留第一次出现的
  });
  return [...map.values()];
}

// 用法
uniqueBy(list, item => item.id);
uniqueBy(list, item => `${item.orgId}-${item.type}`); // 多字段

面试时怎么讲

基本类型首选 Set，一行代码、O(n)。要兼容老环境用 filter + indexOf，但它是 O(n²)，而且 NaN !== NaN 会把 NaN 全留下。对象数组不能直接 Set，因为引用不同，得按业务字段建 Map 去重，通常保留第一次出现的，多字段就拼成字符串当 key。

追问

追问 答
Set 为什么能去重？ 内部用 SameValueZero 比较，NaN 也相等，所以能去重；对象按引用比，两个内容一样的对象仍算不同。
对象数组为什么 Set 不行？ Set 比的是引用，两个 {id:1} 不是同一个对象。
保留最后一次出现的怎么写？ Map 的 set 直接覆盖就行——先遍历、后出现覆盖前面。
大数组性能？ 基本类型 Set 是 O(n)；indexOf 是 O(n²)，几万条就明显慢。
怎么保持原数组不变？ 上面所有实现都返回新数组，没动原数组。

常见错误

  • 以为 [...new Set(objArr)] 能把内容相同的对象去掉（去不掉）。
  • filter + indexOf 处理含 NaN 的数组，结果不对。
  • 多字段 key 用 + 拼接，1-23 和 12-3 撞了，要用分隔符。

6. 顺手也要会（短题）

6.1 手写 call / apply / bind

Function.prototype.myCall = function (ctx, ...args) {
  ctx = ctx == null ? globalThis : Object(ctx);
  const key = Symbol('fn');
  ctx[key] = this;
  const res = ctx[key](...args);
  delete ctx[key];
  return res;
};

Function.prototype.myApply = function (ctx, args = []) {
  return this.myCall(ctx, ...args);
};

Function.prototype.myBind = function (ctx, ...args) {
  const fn = this;
  return function bound(...rest) {
    // new 调用时 this 要指向新对象，而不是绑定的 ctx
    return fn.apply(this instanceof bound ? this : ctx, [...args, ...rest]);
  };
};

6.2 手写 new

function myNew(Ctor, ...args) {
  const obj = Object.create(Ctor.prototype);
  const res = Ctor.apply(obj, args);
  return res && (typeof res === 'object' || typeof res === 'function') ? res : obj;
}

6.3 手写 instanceof

function myInstanceof(obj, Ctor) {
  let proto = Object.getPrototypeOf(obj);
  while (proto) {
    if (proto === Ctor.prototype) return true;
    proto = Object.getPrototypeOf(proto);
  }
  return false;
}

6.4 数组扁平化

function flatten(arr, depth = Infinity) {
  if (depth <= 0) return arr.slice();
  return arr.reduce((acc, cur) =>
    acc.concat(Array.isArray(cur) ? flatten(cur, depth - 1) : cur), []);
}
// 或者：arr.flat(Infinity)

6.5 手写 Promise.all

function myAll(list) {
  return new Promise((resolve, reject) => {
    const arr = Array.from(list);
    const results = new Array(arr.length);
    let count = 0;
    if (!arr.length) return resolve(results);

    arr.forEach((p, i) => {
      Promise.resolve(p).then(
        (v) => {
          results[i] = v;
          if (++count === arr.length) resolve(results);
        },
        reject
      );
    });
  });
}

6.6 柯里化与组合

function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn.apply(this, args);
    return (...rest) => curried.apply(this, [...args, ...rest]);
  };
}

const compose = (...fns) => x => fns.reduceRight((v, f) => f(v), x);
const pipe = (...fns) => x => fns.reduce((v, f) => f(v), x);

6.7 发布订阅

class EventEmitter {
  constructor() { this.events = new Map(); }
  on(name, fn) {
    if (!this.events.has(name)) this.events.set(name, []);
    this.events.get(name).push(fn);
    return this;
  }
  off(name, fn) {
    const list = this.events.get(name);
    if (list) this.events.set(name, list.filter(f => f !== fn));
    return this;
  }
  once(name, fn) {
    const wrapper = (...args) => { fn(...args); this.off(name, wrapper); };
    return this.on(name, wrapper);
  }
  emit(name, ...args) {
    (this.events.get(name) || []).slice().forEach(fn => fn(...args));
    return this;
  }
}

6.8 LRU 缓存（Map 能保插入顺序）

class LRUCache {
  constructor(capacity) {
    this.capacity = capacity;
    this.map = new Map();
  }
  get(key) {
    if (!this.map.has(key)) return -1;
    const val = this.map.get(key);
    this.map.delete(key);
    this.map.set(key, val);   // 重新插入 = 移到最新
    return val;
  }
  put(key, val) {
    if (this.map.has(key)) this.map.delete(key);
    this.map.set(key, val);
    if (this.map.size > this.capacity) {
      this.map.delete(this.map.keys().next().value); // 删最旧的
    }
  }
}

6.9 异步重试

function retry(fn, times = 3, delay = 500) {
  return new Promise((resolve, reject) => {
    const attempt = (left) => {
      Promise.resolve()
        .then(fn)
        .then(resolve)
        .catch(err => {
          if (left <= 1) return reject(err);
          const wait = delay * (times - left + 1); // 递增等待
          setTimeout(() => attempt(left - 1), wait);
        });
    };
    attempt(times);
  });
}

function formatMoney(num, decimals = 2) {

7. 写题时的加分细节

  • 主动说复杂度：去重 Set 是 O(n)，indexOf 是 O(n²)；并发控制是 O(n) 任务本身的开销。
  • 主动说边界：空数组、同步任务、任务抛错、this 丢失。
  • 能挂到项目上就挂：防抖说搜索、节流说滚动、深拷贝说缓存快照、并发控制说批量请求。
  • 别背实现：手写题考的是"你能不能想清楚"，卡住时说思路，比默写完整代码得分高。
